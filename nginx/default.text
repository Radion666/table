server {
    listen 80;
    server_name localhost;

    # Обработка запросов к API


    root /usr/share/nginx/html;  # Путь к статическим файлам фронтенда.
    index index.html;  # Указание файла по умолчанию.

    # Обработка всех остальных запросов
    location / {
       

        # Попытка найти файл или каталог, если не найдено, перенаправляем на index.html
        try_files $uri /index.html$is_args$args =404; 
    }

    # Обработка ошибок
    error_page 500 502 503 504 /50x.html;
    location = /50x.html {
        root /usr/share/nginx/html;  # Путь к странице ошибки.
    }
}
